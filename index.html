function submitRequest() { const input = document.getElementById('requestInput'); const value = input.value.trim(); if (!value) { alert('グループ名を入力してください。'); return; } const gasUrl = "https://script.google.com/macros/s/AKfycbyTMJGRFka6NZq9xokH1KwCt74xkx5yoiW8KQmrcNny0CVohQp_723zYXNLIXliUMnDUA/exec"; const msgElement = document.getElementById('requestMsg'); msgElement.textContent = "送信中..."; msgElement.classList.remove('hidden'); const formData = new URLSearchParams(); formData.append("groupName", value); fetch(gasUrl, { method: "POST", mode: "no-cors", body: formData }) .then(() => { msgElement.textContent = "リクエストを送信しました！確認後に反映されます。"; input.value = ''; setTimeout(() => { msgElement.classList.add('hidden'); }, 5000); }) .catch(err => { alert('送信に失敗しました。'); msgElement.classList.add('hidden'); }); }
